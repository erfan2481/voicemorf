name: Generate Keystore

on:
  workflow_dispatch: {}

jobs:
  generate:
    runs-on: ubuntu-latest
    steps:
      - name: Setup JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Generate keystore
        run: |
          keytool -genkeypair -v \
            -keystore release.keystore \
            -alias voicemorfkey \
            -keyalg RSA -keysize 2048 -validity 10000 \
            -storepass "${{ secrets.KEYSTORE_PASSWORD }}" \
            -keypass "${{ secrets.KEY_PASSWORD }}" \
            -dname "CN=VoiceMorf, OU=Dev, O=VoiceMorf, L=Tehran, S=Tehran, C=IR"

      - name: Self-verify with the SAME secrets
        run: |
          keytool -list -v \
            -keystore release.keystore \
            -alias voicemorfkey \
            -storepass "${{ secrets.KEYSTORE_PASSWORD }}" \
            -keypass "${{ secrets.KEY_PASSWORD }}"
          echo "✅ keystore verified OK with current secrets"

      - name: Encode for secret
        run: base64 -w 0 release.keystore > release.keystore.base64.txt

      - name: Upload keystore artifacts
        uses: actions/upload-artifact@v4
        with:
          name: voicemorf-keystore
          path: |
            release.keystore
            release.keystore.base64.txt
